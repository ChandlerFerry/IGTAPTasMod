	private void Update()
	{
		if (!pauseMenu.menuOpen && !GamePaused)
		{
			if (dashAction.WasPressedThisFrame() || dashAction2.WasPressedThisFrame())
			{
				dashBuffer = true;
				CancelInvoke("cancelDashBuffer");
				Invoke("cancelDashBuffer", 0.12f);
			}
			if (jumpAction.WasPressedThisFrame() || jumpAction2.WasPressedThisFrame())
			{
				jumpBuffer = true;
				CancelInvoke("cancelJumpBuffer");
				CancelInvoke("cancelJumpCutBuffer");
				cancelJumpCutBuffer();
				if (cutsceneMode == cutsceneModes.dash)
				{
					Invoke("cancelJumpBuffer", 0.2f);
				}
				else
				{
					Invoke("cancelJumpBuffer", 0.12f);
				}
			}
			if (jumpAction.WasReleasedThisFrame() || jumpAction2.WasReleasedThisFrame())
			{
				jumpCutBuffer = true;
				CancelInvoke("cancelJumpCutBuffer");
				Invoke("cancelJumpCutBuffer", 0.26f);
			}
			if (resetAction.WasPressedThisFrame() && !GamePaused)
			{
				respawn(manuallyTriggered: true);
			}
		}
		HandleAfterimages();
		float num = 10f;
		float num2 = 0f;
		if (cutsceneMode == cutsceneModes.dash)
		{
			num2 = 2f;
			num += 10f;
		}
		Vector2 vector;
		Vector2 vector2;
		Vector2 origin;
		Vector2 origin2;
		if (facingRight)
		{
			vector = base.transform.position + new Vector3(24f, num);
			vector2 = base.transform.position + new Vector3(27f + num2, num);
			origin = base.transform.position + new Vector3(15f, 19f);
			origin2 = base.transform.position + new Vector3(15f, -18f);
		}
		else
		{
			vector = base.transform.position + new Vector3(-24f, num);
			vector2 = base.transform.position + new Vector3(-27f - num2, num);
			origin = base.transform.position + new Vector3(-15f, 19f);
			origin2 = base.transform.position + new Vector3(-15f, -18f);
		}
		Debug.DrawRay(vector, wallJumpFacing * (6f + num2), Color.red);
		Debug.DrawRay(vector2, Vector2.down * (33f + num), Color.red);
		RaycastHit2D raycastHit2D = Physics2D.Raycast(vector, wallJumpFacing, 6f + num2, groundRaycastLayer);
		RaycastHit2D raycastHit2D2 = Physics2D.Raycast(vector2, Vector2.down, 33f + num, groundRaycastLayer);
		RaycastHit2D raycastHit2D3 = Physics2D.Raycast(origin2, wallJumpFacing, 14f, groundRaycastLayer);
		RaycastHit2D raycastHit2D4 = Physics2D.Raycast(origin, wallJumpFacing, 14f, groundRaycastLayer);
		RaycastHit2D raycastHit2D5 = ((!onGround && !dashActive) ? raycastHit2D3 : raycastHit2D);
		if (dashFramesRemaining != dashFrames && (bool)raycastHit2D2 && !raycastHit2D5 && !raycastHit2D4 && Mathf.Abs(Velocity.x) > runSpeed * 10f / 2f && body.linearVelocityY < 10f)
		{
			Vector2 vector3 = default(Vector2);
			vector3.y = raycastHit2D2.point.y + defaultColliderSize.y + defaultColliderOffset.y;
			if (vector3.y > base.transform.position.y)
			{
				Vector3 position = base.transform.position;
				base.transform.position = new Vector2(base.transform.position.x, vector3.y - 38f);
				stepUpAnim.onSteppedUp(position - base.transform.position);
			}
		}
		if (!WaitingForDashInput)
		{
			return;
		}
		MoveAxis = NormaliseInput(moveAction.ReadValue<Vector2>());
		if (MoveAxis != Vector2.zero && (omniDashUnlocked || MoveAxis.x != 0f))
		{
			dashDirection = MoveAxis;
			if (!omniDashUnlocked)
			{
				dashDirection.y = 0f;
			}
			if (onGround && dashDirection.y < 0f)
			{
				dashDirection.y = 0f;
			}
			dashDirection.Normalize();
		}
	}

	private void FixedUpdate()
	{

			{
				framesSquished++;

		RaycastHit2D raycastHit2D12 = Physics2D.Raycast(vector7, wallJumpFacing, 12f + num, blockingGroundRaycastLayer);

				{

				}
				else

					touchedMossyGround(touchedGround: false);
				}

			bool flag = false;

				normalCollider.size = defaultColliderSize;

					DashBounceFrames = 4;

					dashDirection.Normalize();

					dashFramesRemaining += 3f;
					canHyper = true;
				}
			}
			else if (dashDirection.y > 0f && dashDirection.x != 0f && (raycastHit2D10.collider != null || raycastHit2D11.collider != null) && wallJumpUnlocked)
			{
				dashDirection.x = 0f;

			Vector2 vector8 = dashDirection;
			if (dashDirection.y > 0f)

		if (momentum.x > 0f)
		{
			momentum -= new Vector2(flatMomentumDecay, 0f);

		}
		CalcMovingPlatformVelocity();
		if (!squishedThisFrame)
		{
			body.linearVelocity = Velocity + momentum + movingPlatformVelocity;
		}
		else
		{
			body.linearVelocity = movingPlatformVelocity;
		}
	}

	private void ApplyWallslideFriction(float speedMult)
	{
		if (cutsceneMode != cutsceneModes.longfall && !isFinalWallClingActive)
		{
			if ((double)MoveAxis.y < -0.6)
			{
				speedMult *= 1.5f;
			}
			else if ((double)MoveAxis.y > 0.6)
			{
				speedMult *= 0.35f;
			}
			if (wallJumpsLeft > 0)
			{
				Velocity.y += 50f * Time.fixedDeltaTime * (gravity - (float)wallSlideFriction);
				if (Velocity.y < (float)(-maxWallSlideSpeed) * speedMult * 0.8f)
				{
					Velocity.y = Mathf.Lerp(Velocity.y, (float)(-maxWallSlideSpeed) * speedMult, PostMaxDownwardsSpeedDrag);
				}
			}
			else
			{
				Velocity.y += 50f * Time.fixedDeltaTime * (gravity - (float)(wallSlideFriction + 2));
			}
		}
		if (!OnWall && wallJumpsLeft > 0)
		{
			Velocity.y = Mathf.Clamp(Velocity.y, -150f * speedMult, 0f);
		}
	}

	private void CalcMovingPlatformVelocity()
	{
		if (currentlyTouchedMovingPlatform != null && (onGround || OnWall))
		{
			movingPlatformCoyote = maxWallCoyoteFrames;
			if (OnWall)
			{
				Velocity.y = Mathf.Lerp(Velocity.y, 0f, 0.2f);
			}
			movingPlatformVelocity = currentlyTouchedMovingPlatform.linearV;
			if (platformMomentumHistory.Count == movingPlatformMomentumCoyoteFrames)
			{
				platformMomentumHistory.Dequeue();
			}
			platformMomentumHistory.Enqueue(movingPlatformVelocity);
			if (movingPlatformVelocity.y < 0f && !OnWall)
			{
				movingPlatformVelocity *= new Vector2(1f, 1.2f);
			}
			if (OnWall)
			{
				if ((base.gameObject.transform.localScale.x < 0f && movingPlatformVelocity.x < 0f) || (base.gameObject.transform.localScale.x > 0f && movingPlatformVelocity.x > 0f))
				{
					movingPlatformVelocity *= new Vector2(2f, 1f);
				}
				else
				{
					movingPlatformVelocity *= new Vector2(0.9f, 1f);
				}
			}
			if (touchingSideOfMovingBlock)
			{
				if ((facingRight && movingPlatformVelocity.x < 0f) || (!facingRight && movingPlatformVelocity.x > 0f))
				{
					movingPlatformVelocity = Vector2.zero;
				}
				else if (onGround)
				{
					movingPlatformVelocity = Vector2.zero;
				}
			}
		}
		else
		{
			if (movingPlatformCoyote > -1f)
			{
				movingPlatformCoyote -= 1f;
			}
			movingPlatformVelocity = Vector2.zero;
			if (platformMomentumHistory.Count == movingPlatformMomentumCoyoteFrames)
			{
				platformMomentumHistory.Dequeue();
			}
			platformMomentumHistory.Enqueue(movingPlatformVelocity);
		}
	}

	private void CheckForSuperJump(bool isDashBounce = false)
	{
		if (DashBounceFrames > 0 || (canHyper && framesSinceLastDash < 4f) || !jumpBuffer || !onGround)
		{
			return;
		}
		MovingPlatformMomentumInherit(Vector2.zero);
		cutsceneMode = cutsceneModes.none;
		jumpBuffer = false;
		CancelInvoke("cancelJumpBuffer");
		coyoteFrames = 0f;
		onGround = false;
		base.transform.position = (base.transform.position += new Vector3(0f, 2f));
		recentlyJumped = true;
		Invoke("endRecentJump", 0.03f);
		playSFX("bigJump");
		JumpingSquashStretch();
		if (currentGroundType == groundTypes.Metal)
		{
			metalLandParticles.Emit(8);
		}
		else if (currentGroundType == groundTypes.Mossy)
		{
			grassLandParticles.Emit(6);
		}
		normalCollider.offset = dashColliderOffset;
		normalCollider.size = dashColliderSize;
		if (canHyper)
		{
			MostRecentUpwardsMoveWasNormalJump = false;
			Velocity.y = jumpForce * hyperJumpHeightMod;
			if (MoveAxis.x > 0f)
			{
				addMomentum(Vector2.right, hyperJumpMagnitude);
			}
			else if (MoveAxis.x < 0f)
			{
				addMomentum(Vector2.left, hyperJumpMagnitude);
			}
			if (Mathf.Sign(preDashMomentum.x) == Mathf.Sign(dashDirection.x))
			{
				if (base.transform.position.y < lastHyperY - 4f)
				{
					momentum.x += preDashMomentum.x;
					Debug.Log("Did an ultra!");
				}
				TimeStop(0.05f);
			}
			lastHyperY = base.transform.position.y;
		}
		else
		{
			MostRecentUpwardsMoveWasNormalJump = true;
			Velocity.y = jumpForce;
			if (MoveAxis.x > 0f)
			{
				addMomentum(Vector2.right, dashJumpMagnitude);
			}
			else if (MoveAxis.x < 0f)
			{
				addMomentum(Vector2.left, dashJumpMagnitude);
			}
			else
			{
				addMomentum(new Vector2(dashDirection.x, 0f), dashJumpMagnitude);
			}
		}
	}

	public void MovingPlatformMomentumInherit(Vector2 linearV, bool triggeredByJump = true)
	{
		if (currentlyTouchedMovingPlatform == null && movingPlatformCoyote <= 0f)
		{
			return;
		}
		Vector2 vector = Vector2.zero;
		if (triggeredByJump)
		{
			movingPlatformCoyote = -1f;
			float num = -9999999f;
			foreach (Vector2 item in platformMomentumHistory)
			{
				if (item.magnitude > num)
				{
					num = item.magnitude;
					vector = item;
				}
			}
		}
		else
		{
			vector = linearV;
		}
		if (movingPlatformInheritCooldown > 0f)
		{
			if (lastInheritedPlatformV.sqrMagnitude > vector.sqrMagnitude)
			{
				return;
			}
			momentum -= lastInheritedPlatformV;
			lastInheritedPlatformV = Vector2.zero;
		}
		momentum += vector;
		lastInheritedPlatformV = vector;
		movingPlatformInheritCooldown = MaxMovingPlatformInheritCd;
		if (currentlyTouchedMovingPlatform != null)
		{
			if (triggeredByJump)
			{
				currentlyTouchedMovingPlatform.MakeIntangible();
			}
			currentlyTouchedMovingPlatform = null;
		}
		else if (triggeredByJump)
		{
			lastTouchedMovingPlatform.MakeIntangible();
		}
	}

	private Vector2 NormaliseInput(Vector2 MoveAxis)
	{
		if ((double)MoveAxis.x > 0.8)
		{
			MoveAxis.x = 1f;
		}
		else if ((double)MoveAxis.x < -0.8)
		{
			MoveAxis.x = -1f;
		}
		else if ((double)MoveAxis.x > 0.4 && (double)Mathf.Abs(MoveAxis.y) > 0.3)
		{
			MoveAxis.x = 1f;
		}
		else if ((double)MoveAxis.x < -0.4 && (double)Mathf.Abs(MoveAxis.y) > 0.3)
		{
			MoveAxis.x = -1f;
		}
		else if ((double)Mathf.Abs(MoveAxis.x) < 0.3)
		{
			MoveAxis.x = 0f;
		}
		if ((double)MoveAxis.y > 0.4)
		{
			MoveAxis.y = 1f;
		}
		else if ((double)MoveAxis.y < -0.4)
		{
			MoveAxis.y = -1f;
		}
		else
		{
			MoveAxis.y = 0f;
		}
		return MoveAxis;
	}

	private void getPlayerControlledInput()
	{
		if (dashBuffer && dashCooldown < 0f && dashUnlocked && !(wallJumpMovementLockRemaining < wallJumpMovementLockFrames - 5f))
		{
			CancelInvoke("cancelDashBuffer");
			Invoke("cancelDashBuffer", 0.06f);
		}
		if (dashBuffer && dashCooldown < 0f && dashUnlocked && wallJumpMovementLockRemaining < wallJumpMovementLockFrames - 5f)
		{
			if (airDashesLeft > 0)
			{
				if (blockSwapUnlocked)
				{
					swapBlocks();
				}
				dashBuffer = false;
				CancelInvoke("cancelDashBuffer");
				TimeStop(dashTimeStopTime);
				dashFramesRemaining = dashFrames;
				if (MoveAxis == Vector2.zero || (!omniDashUnlocked && MoveAxis.x == 0f))
				{
					dashDirection = new Vector2(facingRight ? 1 : (-1), 0f);
					WaitingForDashInput = true;
				}
				else
				{
					dashDirection = MoveAxis;
					if (Math.Abs(dashDirection.x) < 1f && (double)Math.Abs(dashDirection.y) >= 0.4)
					{
						dashDirection.x = 0f;
					}
					if (!omniDashUnlocked)
					{
						dashDirection.y = 0f;
					}
					if (onGround && dashDirection.y < 0f)
					{
						dashDirection.y = 0f;
					}
					dashDirection.Normalize();
				}
				if (OnWall && wallCoyoteFrames == maxWallCoyoteFrames)
				{
					facingRight = !facingRight;
					dashDirection.x = (facingRight ? 1 : (-1));
					Debug.Log(OnWall);
					Debug.Log(wallCoyoteFrames + " " + maxWallCoyoteFrames);
					wallCoyoteFrames = 0f;
				}
				preDashMomentum = momentum;
				momentum = Vector2.zero;
				if (dashDirection.y > 0f)
				{
					recentlyJumped = true;
					Invoke("endRecentJump", 0.03f);
				}
				framesSinceLastDash = 0f;
				wallJumpDirection = 0f;
				cutsceneMode = cutsceneModes.dash;
				base.transform.position += (Vector3)dashDirection * -20f;
				Velocity = dashDirection * dashSpeed * 10f;
				airDashesLeft--;
				dashActive = true;
				dashCooldown = 100f;
				playSFX("dash");
				wallJumpMovementLockRemaining = -1f;
			}
			else if (currentGroundType == groundTypes.Mossy && tiredSoundCooldown < 0f)
			{
				tiredSoundCooldown = 0.2f;
			}
			return;
		}
		MoveLeftAndRight();
		if (MostRecentUpwardsMoveWasNormalJump && jumpCutBuffer && Velocity.y < 580f && Velocity.y > 150f)
		{
			Velocity.y = 50f;
			if (wallJumpMovementLockRemaining > 0f)
			{
				wallJumpMovementLockRemaining = 0f;
			}
		}
		if (jumpBuffer && onGround)
		{
			MovingPlatformMomentumInherit(Vector2.zero);
			MostRecentUpwardsMoveWasNormalJump = true;
			jumpBuffer = false;
			CancelInvoke("cancelJumpBuffer");
			coyoteFrames = 0f;
			onGround = false;
			base.transform.position = (base.transform.position += new Vector3(0f, 5f));
			Velocity.y = jumpForce;
			recentlyJumped = true;
			Invoke("endRecentJump", 0.03f);
			playSFX("bigJump");
			JumpingSquashStretch();
			if (currentGroundType == groundTypes.Metal)
			{
				metalLandParticles.Emit(4);
			}
			else if (currentGroundType == groundTypes.Mossy)
			{
				grassLandParticles.Emit(3);
			}
			normalCollider.offset = dashColliderOffset;
			normalCollider.size = dashColliderSize;
			if (DashBounceFrames > 0)
			{
				addMomentum(Vector2.up, dashBounceMagnitude);
				addMomentum(new Vector2(1f, 0f), momentum.x * 0.05f);
				MostRecentUpwardsMoveWasNormalJump = false;
			}
		}
		else if (jumpBuffer && (OnWall || wallCoyoteFrames > 0f))
		{
			if (wallJumpsLeft > 0 && wallJumpUnlocked)
			{
				MostRecentUpwardsMoveWasNormalJump = true;
				if (framesSinceLastDash > dashBoostMin && framesSinceLastDash < dashBoostMax)
				{
					SpecialWallJump(0.6f, 0.6f, Vector2.up * dashBoostUpwards);
					return;
				}
				MovingPlatformMomentumInherit(Vector2.zero);
				BoxCollider2D boxCollider2D = normalCollider;
				boxCollider2D.excludeLayers = (int)boxCollider2D.excludeLayers + LayerMask.GetMask("Ground");
				Invoke("reEnableNormalCollider", 0.01f);
				jumpBuffer = false;
				CancelInvoke("cancelJumpBuffer");
				wallJumpsLeft--;
				wallCoyoteFrames = 0f;
				OnWall = false;
				base.transform.position += new Vector3(wallJumpDirection, 1f);
				wallJumpDirection = 0f - base.gameObject.transform.localScale.x;
				Velocity.y = wallJumpForce;
				wallJumpMovementLockRemaining = wallJumpMovementLockFrames;
				Velocity.x = runSpeed * wallJumpDirection * 10f;
				JumpingSquashStretch();
				normalCollider.offset = dashColliderOffset;
				normalCollider.size = dashColliderSize;
				if (currentGroundType == groundTypes.Metal)
				{
					playSFX("smallJump");
					ParticleSystem.EmissionModule emission = metalWallParticles.emission;
					emission.rateOverTime = 0f;
					metalWallParticles.Emit(3);
				}
				else if (currentGroundType == groundTypes.Mossy)
				{
					Singleton<Audio>.Instance.PlayAudio(Audio.AUDIO_ID.GrassFootstep, 1f, 0.15f);
					playSFX("");
					ParticleSystem.EmissionModule emission2 = grassWallParticles.emission;
					emission2.rateOverTime = 0f;
					grassWallParticles.Emit(3);
				}
			}
			else if (wallJumpUnlocked && currentGroundType == groundTypes.Mossy && tiredSoundCooldown < 0f)
			{
				tiredSoundCooldown = 0.2f;
			}
		}
		else
		{
			if (!jumpBuffer || !doubleJumpUnlocked || onGround || !(wallCoyoteFrames < maxWallCoyoteFrames))
			{
				return;
			}
			if (airJumpsLeft > 0)
			{
				if ((MostRecentUpwardsMoveWasNormalJump && Velocity.y < 200f) || (!MostRecentUpwardsMoveWasNormalJump && Velocity.y < 700f))
				{
					MostRecentUpwardsMoveWasNormalJump = true;
					jumpBuffer = false;
					CancelInvoke("cancelJumpBuffer");
					airJumpsLeft--;
					if (Velocity.y < 0f)
					{
						Velocity.y = 0f;
					}
					Velocity.y += airJumpForce;
					Velocity.y = Mathf.Lerp(Velocity.y, airJumpForce, 0.5f);
					playSFX("airJump");
					JumpingSquashStretch();
					normalCollider.offset = dashColliderOffset;
					normalCollider.size = dashColliderSize;
				}
			}
			else if (maxAirJumps > 0 && currentGroundType == groundTypes.Mossy && tiredSoundCooldown < 0f)
			{
				tiredSoundCooldown = 0.2f;
			}
		}
	}

	private void MoveLeftAndRight()
	{
		if (wallJumpMovementLockRemaining < 0f)
		{
			wallJumpDirection = 0f;
			if (momentum.x > 0f && MoveAxis.x < 0f)
			{
				momentum.x -= MovingOppositeMomentumDecay;
			}
			else if (momentum.x < 0f && MoveAxis.x > 0f)
			{
				momentum.x += MovingOppositeMomentumDecay;
			}
			if (MoveAxis.x != 0f)
			{
				Velocity.x += MoveAxis.x * runSpeed * 10f / framesToReachTopSpeed;
				if (Mathf.Sign(Velocity.x) != Mathf.Sign(MoveAxis.x))
				{
					Velocity.x += MoveAxis.x * runSpeed * 10f / framesToReachTopSpeed;
				}
				if (atJumpApex)
				{
					Velocity.x += jumpApexSpeedBonus * MoveAxis.x * 10f;
				}
			}
			else
			{
				if (Velocity.x > 0f)
				{
					Velocity.x -= runSpeed * 10f / framesToReachTopSpeed;
				}
				else if (Velocity.x < 0f)
				{
					Velocity.x += runSpeed * 10f / framesToReachTopSpeed;
				}
				if (Mathf.Abs(Velocity.x) < runSpeed * 10f / framesToReachTopSpeed)
				{
					Velocity.x = 0f;
				}
			}
			if (Velocity.x > runSpeed * 10f)
			{
				Velocity.x = 1f * runSpeed * 10f;
			}
			else if (Velocity.x < (0f - runSpeed) * 10f)
			{
				Velocity.x = -1f * runSpeed * 10f;
			}
		}
		else
		{
			Velocity.x = wallJumpDirection * runSpeed * 10f;
			wallJumpMovementLockRemaining -= 1f;
		}
	}

	private void HandleAfterimages()
	{
		if ((momentum.magnitude > afterimageMomentumMagnitudeRequired && body.linearVelocity.magnitude > 400f) || body.linearVelocityY < 0f - 10f * maxDownwardsSpeed - 1f || dashActive)
		{
			if (dashActive)
			{
				timeSinceLastAfterimage += Time.deltaTime * body.linearVelocity.magnitude;
			}
			else
			{
				timeSinceLastAfterimage += Time.deltaTime * body.linearVelocity.magnitude;
			}
			if (timeSinceLastAfterimage > afterimageFrequency)
			{
				timeSinceLastAfterimage = 0f;
				SpriteRenderer spriteRenderer = inactiveAfterimages.Pop();
				spriteRenderer.sprite = sr.sprite;
				spriteRenderer.gameObject.SetActive(value: true);
				spriteRenderer.transform.position = base.transform.position;
				spriteRenderer.transform.localScale = base.transform.localScale;
				afterimageTimeAlive.Add(0f);
				activeAfterimages.Add(spriteRenderer);
			}
		}
		List<int> list = new List<int>();
		for (int i = 0; i < activeAfterimages.Count; i++)
		{
			afterimageTimeAlive[i] += Time.deltaTime / afterimageLifetime;
			SpriteRenderer spriteRenderer2 = activeAfterimages[i];
			Color color = Color.Lerp(afterimageColors[0], afterimageColors[1], afterimageTimeAlive[i]);
			if (afterimageTimeAlive[i] >= 1f)
			{
				list.Add(i);
			}
			spriteRenderer2.color = color;
		}
		foreach (int item in list)
		{
			try
			{
				activeAfterimages[item].gameObject.SetActive(value: false);
				inactiveAfterimages.Push(activeAfterimages[item]);
				activeAfterimages.RemoveAt(item);
				afterimageTimeAlive.RemoveAt(item);
			}
			catch
			{
			}
		}
		list.Clear();
	}

	private void endDash(bool capVerticalV = true)
	{
		if (!hitDashResetterWhileInCurrentDash)
		{
			if (airDashesLeft == 0)
			{
				dashCooldown = defaultDashCooldown;
			}
			else if (onGround)
			{
				dashCooldown = defaultDashCooldown;
			}
			else
			{
				dashCooldown = 0.02f;
			}
		}
		else
		{
			hitDashResetterWhileInCurrentDash = false;
			dashCooldown = 0.02f;
			TimeStop(0.06f);
		}
		if (dashDirection.y > 0f)
		{
			MostRecentUpwardsMoveWasNormalJump = false;
			if (capVerticalV)
			{
				Velocity *= 0.35f;
			}
		}
		else if (Velocity.y < 10f)
		{
			Velocity.y = 10f;
		}
		if (cutsceneMode == cutsceneModes.dash)
		{
			if ((Mathf.Sign(preDashMomentum.x) == Mathf.Sign(dashDirection.x) && dashDirection.y <= 0f) || dashDirection.y == -1f)
			{
				momentum.x += preDashMomentum.x * 0.95f;
				TimeStop(0.05f);
			}
			if (Mathf.Sign(preDashMomentum.y) == Mathf.Sign(dashDirection.y) && dashDirection.y != 0f)
			{
				momentum.y += preDashMomentum.y * 0.95f;
				TimeStop(0.05f);
			}
		}
		cutsceneMode = cutsceneModes.none;
		dashActive = false;
		canHyper = false;
		wallBouncePossible = false;
		if (OnWall)
		{
			PlayWallSlideSfx(currentGroundType);
		}
	}

	private void LandingSquashStretch()
	{
		StartCoroutine(IE_Land());
		IEnumerator IE_Land()
		{
			Transform animSprite = animator.transform;
			Vector2 initialScale = animSprite.localScale;
			Vector2 targetScale = new Vector2(animSprite.localScale.y * 1.2f, animSprite.localScale.x * 0.9f) * 1f;
			float durationMult = 2f + 60f * (initialScale.y - 0.9f);
			float i = Time.deltaTime * durationMult;
			while (i < 1f)
			{
				i += Time.deltaTime * durationMult;
				animSprite.localScale = Vector2.Lerp(initialScale, targetScale, i);
				animSprite.localPosition = new Vector3(0f, (animSprite.localScale.y - 1f) * 50f);
				yield return new WaitForEndOfFrame();
			}
			i = 0f;
			while (i < 1f)
			{
				i += Time.deltaTime * durationMult;
				animSprite.localScale = Vector2.Lerp(targetScale, Vector2.one, i);
				animSprite.localPosition = new Vector3(0f, (animSprite.localScale.y - 1f) * 50f);
				yield return new WaitForEndOfFrame();
			}
			animSprite.localScale = new Vector2(1f, 1f);
			animSprite.localPosition = Vector2.zero;
		}
	}

	private void FallingSquashStretch()
	{
		if (Velocity.y < 0f)
		{
			float num = 300f;
			float num2 = 0.0028f;
			float num3 = Mathf.Sqrt(Mathf.Max(0f - Velocity.y - num, 0f));
			num3 *= num2;
			animator.transform.localScale = new Vector2(1f - num3, 1f + num3);
		}
	}

	private void JumpingSquashStretch()
	{
		StartCoroutine(IE_Jump());
		IEnumerator IE_Jump()
		{
			Vector2 initialScale = new Vector2(0.78f, 1.19f);
			float i = 0f;
			while (i < 1f)
			{
				i += Time.deltaTime * 8.2f;
				animator.transform.localScale = Vector2.Lerp(initialScale, Vector2.one, i);
				yield return new WaitForEndOfFrame();
			}
			animator.transform.localScale = Vector2.one;
		}
	}

	public void TimeStop(float duration)

	public void hitSpring(float upForce, float strength, float movementLock, Vector2 up)
	{
		if (cutsceneMode == cutsceneModes.dash)
		{
			endDash();
		}
		cutsceneMode = cutsceneModes.spring;
		if (up == Vector2.up || up == Vector2.down)
		{
			cutsceneMode = cutsceneModes.none;
		}
		else
		{
			Invoke("cancelSpringCutscene", movementLock);
		}
		if (OnWall)
		{
			ParticleSystem.EmissionModule emission = metalWallParticles.emission;
			emission.rateOverTime = 0f;
			emission = grassWallParticles.emission;
			emission.rateOverTime = 0f;
			playSFX("");
		}
		ach_boingboing++;
		if (SteamManager.Initialized && ach_boingboing == 10)
		{
			SteamUserStats.SetAchievement("ACH_SPRING");
			SteamUserStats.StoreStats();
			bool pbAchieved = false;
			SteamUserStats.GetAchievement("ACH_SPRING", out pbAchieved);
			Debug.Log("achievement unlocked: " + pbAchieved);
		}
		MostRecentUpwardsMoveWasNormalJump = false;
		onGround = false;
		OnWall = false;
		recentlyJumped = true;
		Invoke("endRecentJump", 0.03f);
		airDashesLeft = maxAirDashes;
		airJumpsLeft = maxAirJumps;
		wallJumpsLeft = maxWallJumps;
		base.transform.position += (Vector3)up * 10f;
		addMomentum(Vector2.up, upForce / 4f);
		addMomentum(new Vector2(up.x * 1.6f, up.y / 4f), strength);
		Velocity = new Vector2(0f, upForce * 9f + up.y * strength * 3f);
		wallJumpDirection = up.x;
		normalCollider.offset = dashColliderOffset;
		normalCollider.size = dashColliderSize;
	}

	public override void load(string folder)
	{
		try
		{
			SaveObject saveObject = JsonUtility.FromJson<SaveObject>(File.ReadAllText(Application.persistentDataPath + folder + "/playerdata.txt"));
			respawnPoint = saveObject._respawnPoint;
			dashUnlocked = saveObject._dashUnlocked;
			maxAirDashes = saveObject._maxDashes;
			wallJumpUnlocked = saveObject._wallJumpUnlocked;
			doubleJumpUnlocked = saveObject._airJumpUnlocked;
			maxAirJumps = saveObject._maxAirJumps;
			blockSwapUnlocked = saveObject._blockSwapUnlocked;
			omniDashUnlocked = saveObject._omniDashUnlocked;
			Singleton<globalStats>.Instance.Zone2toZone1PathOpen = saveObject._zone2cleared;
			if (blockSwapUnlocked)
			{
				blockSwapper.swapBlocks(BlueActive: true);
			}
			else
			{
				blockSwapper.swapBlocks(saveObject._isBlueActive);
			}
			Singleton<globalStats>.Instance.currentA1State = saveObject._area1State;
			Singleton<globalStats>.Instance.visitedTeleporters = saveObject._visitedTeleporters;
			for (int i = 0; i < saveObject._globalUpgradeDict.Count; i++)
			{
				SerializedUpgradeDict serializedUpgradeDict = saveObject._globalUpgradeDict[i];
				if (globalStats.globalUpgradeDict.ContainsKey(serializedUpgradeDict._Key))
				{
					globalStats.globalUpgradeDict[serializedUpgradeDict._Key] = serializedUpgradeDict._Value;
				}
			}
			for (int j = 0; j < saveObject._currencies.Count; j++)
			{
				SerializedCurrencyDict serializedCurrencyDict = saveObject._currencies[j];
				if (globalStats.currencyLookup.ContainsKey(serializedCurrencyDict._Key))
				{
					globalStats.currencyLookup[serializedCurrencyDict._Key] = serializedCurrencyDict._Value;
				}
			}
			Singleton<globalStats>.Instance.UpdateVmanScore(saveObject._vmanScore);
			if (!isOnMainMenu)
			{
				cam.setup(respawnPoint, saveObject._camSize);
				Singleton<ZoneLoader>.Instance.activeZone = saveObject._currentZone;
				if (Singleton<ZoneLoader>.Instance.activeZone == 0)
				{
					Singleton<ZoneLoader>.Instance.activeZone = 1;
				}
				init = false;
			}
		}
		catch (Exception ex)
		{
			Debug.Log("Player loading failed! " + ex);
		}
	}
